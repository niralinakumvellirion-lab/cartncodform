  Notes (observations only):
  * Still a Polaris <Page title="COD Orders"> + <Layout> + <Grid> + <IndexTable>
    screen (NOT the div-based redesign Today/Customers/Messages/Insights got).
    One "Refresh" secondaryAction -> loadOrders.
  * Data: GET /api/stores/:shop/orders on mount / refresh. The endpoint returns
    a BARE ARRAY (see routes/stores.js); line 31 also tolerates { orders: [...] }.
  * 3 stat tiles (useMemo over `orders`):
      Total orders      = orders.length
      Pending           = count where status === 'pending'
      Confirmed revenue = sum of productPrice*quantity where status === 'confirmed'
    NOTE the stat computes "confirmed" but the CodOrder default status from
    intake is 'pending' and updateStatus sends 'confirmed' | 'cancelled'.
  * IndexTable, selectable=false, 8 columns:
      Customer  o.name (bold) + o.address (subdued)
      Phone     o.phone
      Product   o.productName + "Qty: {o.quantity}"
      Value     rupee + productPrice*quantity (en-IN)
      Email     o.email or em-dash
      Status    Badge, tone confirmed=success / cancelled=critical / else default
      Date      o.createdAt -> toLocaleDateString('en-IN') or em-dash
      Action    only when status === 'pending': "Confirm" (primary) and
                "Cancel" (critical) buttons -> updateStatus(o._id, ...)
  * updateStatus -> PATCH /api/cod/order/:id { status }, then reload. No
    optimistic update; errors set the top banner.
  * No filter / search / pagination. Whole list rendered.
  * "Customer" column prints o.name + o.address in plain view (PII visible in
    the admin UI — expected for an order-management screen; not logged).

 87	/**
 88	 * PATCH /api/cod/order/:id
 89	 * Owner updates the status of a COD order (pending/confirmed/cancelled).
 90	 */
 91	router.patch('/order/:id', requireAuth, async (req, res) => {
 92	  try {
 93	    const { status } = req.body;
 94	    const allowed = ['pending', 'confirmed', 'cancelled'];
 95
 96	    if (!allowed.includes(status)) {
 97	      return res.status(400).json({ error: `status must be one of ${allowed.join(', ')}` });
 98	    }
 99
100	    const existingOrder = await CodOrder.findById(req.params.id);
101	    if (!existingOrder) {
102	      return res.status(404).json({ error: 'COD order not found' });
103	    }
104
105	    if (req.shopDomain !== existingOrder.shopDomain) {
106	      return res.status(403).json({ error: 'Forbidden' });
107	    }
108
109	    const order = await CodOrder.findByIdAndUpdate(
110	      req.params.id,
111	      { status },
112	      { new: true }
113	    );
114
115	    console.log(`[cod] Order ${order._id} status -> ${status}`);
116	    return res.json({ success: true, order });
117	  } catch (err) {
118	    console.error('[cod] PATCH /order/:id error:', err.message);
119	    return res.status(500).json({ error: 'Failed to update COD order' });
120	  }
121	});
