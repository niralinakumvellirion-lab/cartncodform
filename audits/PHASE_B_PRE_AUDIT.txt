    4   * Identity resolution + profile upsert.
    5   *
    6   *   upsertProfile(shopDomain, identifiers, updates = {})
    7   *
    8   * `identifiers` may contain any subset of:
    9   *   { customerId, email, phone, sessionId, cartToken, pushToken }
   10   *   - sessionId  -> stored in identifiers.sessionIds (ccfSessionId, a stable UUID)
   11   *   - cartToken  -> stored in identifiers.cartTokens
   12   *   - pushToken  -> stored in identifiers.pushTokens (FCM token)
   13   *
   14   * `updates` may mix:
   15   *   - plain dot-notation fields   -> applied via $set   (e.g. 'lastSeenAt', 'channels.push.subscribed')
   16   *   - MongoDB update operators    -> merged at top level (e.g. { $inc: { 'orders.count': 1 } })
   17   *
   18   * Merge priority (highest first): customerId -> email -> phone -> cartToken -> sessionId -> pushToken
   19   *
   20   * Fire-and-forget from every ingest path. Never logs PII — only profileId + shopDomain.
   21   *
   22   * @returns {Promise<Profile|null>} the resolved profile, or null if no usable identifier was supplied.
   23   */

   27  const ARRAY_FIELD = {
   28    email: 'emails',
   29    phone: 'phones',

   35  function cleanIdentifiers(identifiers = {}) {

   38      let val = identifiers[key];
   39      if (val == null) continue;
   40      val = String(val).trim();
   41      if (!val) continue;

   49  function splitUpdates(updates = {}) {
   50    const set = {};
   51    const operators = {};
   52    for (const [key, value] of Object.entries(updates)) {

   54        operators[key] = { ...(operators[key] || {}), ...value };

   56        set[key] = value;
   57      }
   58    }

   62  // Build the $or clause (+ shopDomain) that finds every profile any provided

   64  function buildQuery(shopDomain, ids) {
   65    const or = [];

   76  // Rank a profile by the highest-priority identifier it already contains.
   77  // Lower number = higher priority. PRIORITY.length = matches nothing.
   78  function matchRank(profile, ids) {
   79    const pid = profile.identifiers || {};
   80    for (let i = 0; i < PRIORITY.length; i++) {
   81      const key = PRIORITY[i];
   82      const val = ids[key];
   83      if (!val) continue;
   84      if (key === 'customerId') {
   85        if (pid.customerId && pid.customerId === val) return i;
   86      } else {
   87        const arr = pid[ARRAY_FIELD[key]] || [];
   88        if (arr.includes(val)) return i;
   89      }
   90    }
   91    return PRIORITY.length;
   92  }

   94  // Derive stage from identity + order history, never downgrading.
   95  const STAGE_RANK = { anonymous: 0, identified: 1, lapsed: 2, customer: 3 };
   96  function nextStage(profile) {
   97    const pid = profile.identifiers || {};
   98    const hasCustomer = Boolean(pid.customerId) || (profile.orders && profile.orders.count > 0);
   99    const hasContact = (pid.emails && pid.emails.length > 0) || (pid.phones && pid.phones.length > 0);
  100

  102    if (hasCustomer) candidate = 'customer';
  103    else if (hasContact) candidate = 'identified';

  105    const current = profile.stage || 'anonymous';
  106    return STAGE_RANK[candidate] > STAGE_RANK[current] ? candidate : current;

  128  async function applyToProfile(profileId, shopDomain, ids, updates, existingCustomerId) {
  129    const { set, operators } = splitUpdates(updates);
  130  
  131    const update = { ...operators };
  132    update.$set = { ...(operators.$set || {}), ...set, updatedAt: new Date() };
  133  
  134    // customerId is a scalar — only set it if the profile doesn't already have one.
  135    if (ids.customerId && !existingCustomerId) {
  136      update.$set['identifiers.customerId'] = ids.customerId;
  137    }
  138  
  139    const addToSet = addToSetForIds(ids);
  140    if (Object.keys(addToSet).length > 0) {
  141      update.$addToSet = { ...(operators.$addToSet || {}), ...addToSet };
  142    }
  143  
  144    let doc = await Profile.findOneAndUpdate({ _id: profileId, shopDomain }, update, { new: true });
  145    if (!doc) return null;
  146  
  147    const stage = nextStage(doc);
  148    const capped = capMessages(doc.messages);
  149    const needsMessageCap = Array.isArray(doc.messages) && doc.messages.length > capped.length;
  150  
  151    if (stage !== doc.stage || needsMessageCap) {
  152      doc = await Profile.findOneAndUpdate(
  153        { _id: profileId, shopDomain },
  154        { $set: { stage, messages: capped, updatedAt: new Date() } },
  155        { new: true }
  156      );
  157    }
  158    return doc;
  159  }

  161  async function upsertProfile(shopDomain, identifiers = {}, updates = {}) {
  162    try {
  163      const shop = String(shopDomain || '').trim().toLowerCase();
  164      const ids = cleanIdentifiers(identifiers);
  165  
  166      if (!shop || Object.keys(ids).length === 0) {
  167        // Nothing to resolve on — silently no-op (no PII, no id to log).
  168        return null;
  169      }
  170  
  171      const matches = await Profile.find(buildQuery(shop, ids));
  172  
  173      // ---- 0 matches: create fresh, then apply updates ----
  174      if (matches.length === 0) {
  175        const seed = { shopDomain: shop, identifiers: {} };
  176        if (ids.customerId) seed.identifiers.customerId = ids.customerId;
  177        for (const [key, val] of Object.entries(ids)) {
  178          if (key === 'customerId') continue;
  179          seed.identifiers[ARRAY_FIELD[key]] = [val];
  180        }
  181        const created = await Profile.create(seed);
  182        const result = await applyToProfile(created._id, shop, ids, updates, ids.customerId || null);
  183        console.log(`[profile] created ${result?._id} for ${shop}`);
  184        return result;
  185      }
  186  
  187      // ---- 1 match: add new identifiers + apply updates ----
  188      if (matches.length === 1) {
  189        const p = matches[0];
  190        const result = await applyToProfile(
  191          p._id,
  192          shop,
  193          ids,
  194          updates,
  195          p.identifiers && p.identifiers.customerId
  196        );
  197        return result;
  198      }
  199  
  200      // ---- multiple matches: merge into the highest-priority profile ----
  201      const ranked = matches
  202        .map((p) => ({ p, rank: matchRank(p, ids) }))
  203        .sort((a, b) => a.rank - b.rank || new Date(a.p.createdAt || 0) - new Date(b.p.createdAt || 0));
  204  
  205      const winner = ranked[0].p;
  206      const losers = ranked.slice(1).map((r) => r.p);
  207  
  208      // Collect identifier values from every loser.
  209      const mergedArrays = { emails: [], phones: [], sessionIds: [], cartTokens: [], pushTokens: [] };
  210      let mergedCustomerId = (winner.identifiers && winner.identifiers.customerId) || null;
  211      let mergedMessages = Array.isArray(winner.messages) ? winner.messages.slice() : [];
  212  
  213      for (const loser of losers) {
  214        const lid = loser.identifiers || {};
  215        for (const arrKey of Object.keys(mergedArrays)) {
  216          if (Array.isArray(lid[arrKey])) mergedArrays[arrKey].push(...lid[arrKey]);
  217        }
  218        if (!mergedCustomerId && lid.customerId) mergedCustomerId = lid.customerId;
  219        if (Array.isArray(loser.messages)) mergedMessages.push(...loser.messages);
  220      }
  221  
  222      const mergeUpdate = { $addToSet: {}, $set: { updatedAt: new Date() } };
  223      for (const [arrKey, vals] of Object.entries(mergedArrays)) {
  224        if (vals.length > 0) mergeUpdate.$addToSet[`identifiers.${arrKey}`] = { $each: vals };
  225      }
  226      if (mergedCustomerId && !(winner.identifiers && winner.identifiers.customerId)) {
  227        mergeUpdate.$set['identifiers.customerId'] = mergedCustomerId;
  228      }
  229      mergeUpdate.$set.messages = capMessages(mergedMessages);
  230      if (Object.keys(mergeUpdate.$addToSet).length === 0) delete mergeUpdate.$addToSet;
  231  
  232      await Profile.findOneAndUpdate({ _id: winner._id, shopDomain: shop }, mergeUpdate);
  233      await Profile.deleteMany({ _id: { $in: losers.map((l) => l._id) } });
  234      console.log(`[profile] merged ${losers.length} profile(s) into ${winner._id} for ${shop}`);
  235  
  236      // Re-apply this call's own identifiers + updates onto the winner.
  237      const result = await applyToProfile(
  238        winner._id,
  239        shop,
  240        ids,
  241        updates,
  242        mergedCustomerId
  243      );
  244      return result;
  245    } catch (err) {
  246      console.error('[profile] upsertProfile error:', err.message);
  247      return null;
  248    }

  142  // --- Automation: scheduled job sender -----------------------------------
  143  /**
  144   * Returns true if sending a push right now would violate quiet
  145   * hours (10pm-8am) in the given IANA timezone.
  146   */
  147  function isQuietHours(timezone) {
  148    try {
  149      const now = new Date();
  150      const hour = parseInt(
  151        new Intl.DateTimeFormat('en-US', {
  152          hour: 'numeric',
  153          hour12: false,
  154          timeZone: timezone || 'Asia/Kolkata',
  155        }).format(now),
  156        10
  157      );
  158      return hour >= 22 || hour < 8;
  159    } catch (err) {
  160      console.error('[automation] isQuietHours error:', err.message);
  161      return false; // fail open — don't block sends on a bad timezone
  162    }
  163  }
  164  
  165  /**
  166   * Returns true if this cartToken/customerId has already received
  167   * 3+ automation pushes in the last 24 hours.
  168   */
  169  async function isOverFrequencyCap(shopDomain, cartToken, customerId) {
  170    const since = new Date(Date.now() - 24 * 60 * 60 * 1000);
  171    const orClauses = [];
  172    if (cartToken) orClauses.push({ cartToken });
  173    if (customerId) orClauses.push({ customerId });
  174    if (!orClauses.length) return false;
  175  
  176    const count = await ScheduledJob.countDocuments({
  177      shopDomain,
  178      status: 'sent',
  179      sentAt: { $gte: since },
  180      $or: orClauses,
  181    });
  182    return count >= 3;
  183  }

  185  /**
  186   * Polls for due ScheduledJob rows and sends them. Runs every 30s.
  187   * Uses a claim-update (findOneAndUpdate with status:'pending' filter)
  188   * so this is safe even if multiple instances ever run.
  189   */
  190  async function processScheduledJobs() {
  191    try {
  192      const now = new Date();
  193      const dueJobs = await ScheduledJob.find({
  194        status: 'pending',
  195        runAt: { $lte: now },
  196      }).limit(20);
  197  
  198      for (const job of dueJobs) {
  199        // Check quiet hours + frequency cap BEFORE claiming — if
  200        // blocked, leave the job pending and re-check next tick
  201        // (quiet hours) or skip it permanently (frequency cap).
  202        const Store = require('./models/Store');
  203        const store = await Store.findOne({ shopDomain: job.shopDomain }).select('timezone').lean();
  204        const timezone = store?.timezone || 'Asia/Kolkata';
  205  
  206        if (isQuietHours(timezone)) {
  207          console.log(`[automation] Job ${job._id} deferred — quiet hours (${timezone})`);
  208          continue; // leave pending, retry next tick
  209        }
  210  
  211        const overCap = await isOverFrequencyCap(job.shopDomain, job.cartToken, job.customerId);
  212        if (overCap) {
  213          await ScheduledJob.findOneAndUpdate(
  214            { _id: job._id, status: 'pending' },
  215            { status: 'skipped', error: 'Frequency cap reached (3/24h)' }
  216          );
  217          console.log(`[automation] Job ${job._id} skipped — frequency cap reached`);
  218          continue;
  219        }
  220  
  221        // Claim the job — only proceed if we successfully flip it from pending.
  222        const claimed = await ScheduledJob.findOneAndUpdate(
  223          { _id: job._id, status: 'pending' },
  224          { status: 'sent', sentAt: new Date() },
  225          { new: true }
  226        );
  227        if (!claimed) continue; // another process already claimed it
  228  
  229        try {
  230          const channel = job.channel || 'push';
  231          const payload = job.payload || {};
  232  
  233          if (channel === 'push') {
  234            const baseUrl = payload.url || `https://${job.shopDomain}`;
  235            const urlWithJob = baseUrl + (baseUrl.includes('?') ? '&' : '?') + 'ccf_job=' + job._id.toString();
  236            const result = await sendPushToCustomers(
  237              job.shopDomain,
  238              payload.title || 'You left something behind!',
  239              payload.body || 'Come back and check it out.',
  240              urlWithJob,
  241              payload.imageUrl || null,
  242              false,
  243              job.cartToken || null,
  244              false
  245            );
  246  
  247            if (!result.success || result.sent === 0) {
  248              await ScheduledJob.findByIdAndUpdate(job._id, {
  249                status: 'failed',
  250                error: result.error || 'No active subscriber found',
  251              });
  252              console.log(`[automation] Job ${job._id} failed: no subscriber reached`);
  253            } else {
  254              console.log(`[automation] Job ${job._id} sent successfully`);
  255            }
  256          } else if (channel === 'email') {
  257            const customer = await AbandonedCustomer.findOne({
  258              shopDomain: job.shopDomain,
  259              sessionId: job.cartToken,
  260            });
  261            if (!customer || !customer.email) {
  262              await ScheduledJob.findByIdAndUpdate(job._id, {
  263                status: 'skipped',
  264                error: 'No customer email for this cart',
  265              });
  266              console.log(`[automation] Job ${job._id} skipped — no customer email`);
  267            } else {
  268              const cartUrl = `https://${job.shopDomain}/cart`;
  269              const sendResult = await sendAbandonedCartEmail(customer, {
  270                subject: payload.subject || undefined,
  271                body: payload.body || undefined,
  272                cartUrl,
  273              });
  274              if (!sendResult.success) {
  275                await ScheduledJob.findByIdAndUpdate(job._id, {
  276                  status: 'failed',
  277                  error: sendResult.error || 'Email send failed',
  278                });
  279                console.log(`[automation] Job ${job._id} email failed: ${sendResult.error}`);
  280              } else {
  281                console.log(`[automation] Job ${job._id} email sent successfully`);
  282              }
  283            }
  284          }
  285        } catch (err) {
  286          await ScheduledJob.findByIdAndUpdate(job._id, {
  287            status: 'failed',
  288            error: err.message,
  289          });
  290          console.error(`[automation] Job ${job._id} error:`, err.message);
  291        }
  292      }
  293    } catch (err) {
  294      console.error('[automation] processScheduledJobs error:', err.message);
  295    }
  296  }
  297  
  298  // Poll every 30 seconds.
  299  setInterval(processScheduledJobs, 30 * 1000);
  300  console.log('[automation] Scheduled job poller started (30s interval)');
