  177  async function isOverFrequencyCap(shopDomain, cartToken, customerId) {
  178    const since = new Date(Date.now() - 24 * 60 * 60 * 1000);
  179    const orClauses = [];
  180    if (cartToken) orClauses.push({ cartToken });
  181    if (customerId) orClauses.push({ customerId });
  182    if (!orClauses.length) return false;
  183  
  184    const count = await ScheduledJob.countDocuments({
  185      shopDomain,
  186      status: 'sent',
  187      sentAt: { $gte: since },
  188      $or: orClauses,
  189    });
  190    return count >= 3;
  191  }

  400  function scheduleNightlySignals() {
  401    const now = new Date();
  402    const next2am = new Date();
  403    next2am.setHours(2, 0, 0, 0);
  404    if (next2am <= now) next2am.setDate(next2am.getDate() + 1);
  405    const msUntil2am = next2am - now;
  406    setTimeout(async () => {
  407      try {
  408        const shops = await Store.find({}, 'shopDomain');
  409        for (const s of shops) {
  410          await runNightlySignals(s.shopDomain);
  411        }
  412        // Brain runs after signals are fresh for every shop.
  413        for (const s of shops) {
  414          await runBrainForShop(s.shopDomain)
  415            .catch((err) => console.error('[brain] run error:', err.message));
  416        }
  417  
  418        // Phase F — reset the rolling 7-day push stats for shops whose window
  419        // has aged out (accumulation approximates the window; this closes it).
  420        const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
  421        await Store.updateMany(
  422          { 'pushStats.lastComputedAt': { $lt: sevenDaysAgo } },
  423          {
  424            $set: {
  425              'pushStats.deliveredLast7d': 0,
  426              'pushStats.attemptedLast7d': 0,
  427              'pushStats.rateLast7d': 0,
  428            },
  429          }
  430        ).catch((err) => console.error('[hygiene] stats reset error:', err.message));
  431      } catch (err) {
  432        console.error('[signals] nightly run error:', err.message);
  433      }
  434      scheduleNightlySignals(); // reschedule for next night
