  7	 * POST /api/events
  8	 * Body: { shopDomain, sessionId?, customerId?, token?, events: [...] }
  9	 * Each event: { type, path, pageType, meta, ts }
 10	 * Accepts a batch of events from the storefront JS.
 11	 * Also callable via navigator.sendBeacon (text/plain body).
 12	 */
 13	router.post('/', async (req, res) => {
 14	  try {
 15	    let body = req.body;
 16
 17	    // sendBeacon sends text/plain — parse it manually if needed.
 18	    if (typeof body === 'string') {
 19	      try { body = JSON.parse(body); } catch { return res.status(400).end(); }
 20	    }
 21
 22	    const { shopDomain, sessionId, customerId, token, events } = body;
 23
 24	    if (!shopDomain || !Array.isArray(events) || events.length === 0) {
 25	      return res.status(400).json({ error: 'shopDomain and events[] required' });
 26	    }
 27
 28	    const shop = shopDomain.trim().toLowerCase();
 29	    const docs = events.map(e => ({
 30	      shopDomain: shop,
 31	      sessionId:  sessionId  || undefined,
 32	      customerId: customerId || undefined,
 33	      token:      token      || undefined,
 34	      type:       e.type     || 'unknown',
 35	      path:       e.path     || undefined,
 36	      pageType:   e.pageType || undefined,
 37	      meta:       e.meta     || undefined,
 38	      ts:         e.ts ? new Date(e.ts) : new Date(),
 39	    }));
 40
 41	    await StorefrontEvent.insertMany(docs, { ordered: false });
 42	    console.log(`[events] Saved ${docs.length} event(s) for ${shop}`);
 43
 44	    // Phase C2 — on-ingest identity resolution + signal recompute. Fire-and-
 45	    // forget: the beacon response must not wait on this.
 46	    {
 47	      const { upsertProfile } = require('../services/profileService');
 48	      const { computeSignalsForProfile } = require('../services/signalEngine');
 49	      upsertProfile(shop, {
 50	        sessionId: sessionId || null,
 51	        customerId: customerId || null,
 52	        pushToken: token || null,
 53	      }, { lastSeenAt: new Date() })
 54	        .then((profile) => {
 55	          if (profile?._id) {
 56	            computeSignalsForProfile(profile._id, shop)
 57	              .catch((err) => console.error('[signals] events ingest error:', err.message));
 58	          }
 59	        })
 60	        .catch((err) => console.error('[profile] events upsert error:', err.message));
 61	    }
 62
 63	    // Phase F — a page_view from a suppressed profile lifts push suppression.
 64	    // Fully fire-and-forget: the beacon response goes out immediately.
 65	    if (sessionId && events.some((e) => e.type === 'page_view')) {
 66	      const { clearPushSuppression } = require('../services/pushHygiene');
 67	      const Profile = require('../models/Profile');
 68	      Profile.findOne({ shopDomain: shop, 'identifiers.sessionIds': sessionId })
 69	        .then((profile) => {
 70	          if (profile) {
 71	            clearPushSuppression(profile._id, shop)
 72	              .catch((err) => console.error('[hygiene] resubscribe error:', err.message));
 73	          }
 74	        })
 75	        .catch((err) => console.error('[hygiene] resubscribe lookup error:', err.message));
 76	    }
 77
 78	    return res.status(200).json({ success: true });
 79	  } catch (err) {
 80	    console.error('[events] POST /api/events error:', err.message);
 81	    return res.status(500).json({ error: 'Failed to save events' });
 82	  }
 83	});

  Notes (observations only):
  * POST /api/events is PUBLIC (no requireAuth) — protected only by needing a
    shopDomain + non-empty events[]. Accepts JSON or a text/plain sendBeacon
    body (manually JSON.parsed). Returns 200 {success:true}; 400 on bad body;
    500 on insert failure. insertMany is { ordered:false } so one bad doc
    doesn't drop the batch.
  * The 4 GET routes are all requireAuth + requireStoreOwner, shop-scoped
    (trim + lowercase), .limit(200), sorted ts desc (except /summary which is
    an aggregate).
  * There is NO general "list events for the shop" / time-series / insights
    endpoint. The only aggregate is /summary = counts by `type` over the last
    30 days (fixed window, no query params), returned as
    [{ _id: <type>, count: <n> }, ...] sorted count desc.
  * /customer/:sessionId and /ccfsession/:ccfSessionId are the SAME query
    (both match on the `sessionId` field) — the model stores the stable
    analytics UUID in `sessionId`.
  * console.log on line 42 logs only a count + shop domain — no PII.
  * Event types seen in the storefront block (extensions/.../push-notifications
    .liquid) that land here: page_view, product_view, collection_view, search,
    cart_view, reached_checkout, add_to_cart, remove_from_cart, cart_update,
    page_exit, push_prompt_shown, push_prompt_accepted. (Not enforced by the
    route — `type` defaults to 'unknown'.)

  3	const storefrontEventSchema = new mongoose.Schema({
  4	  shopDomain: { type: String, required: true, lowercase: true, index: true },
  5	  sessionId:  { type: String, index: true },   // Shopify cart token
  6	  customerId: { type: String, index: true },   // Shopify customer id (logged-in only)
  7	  token:      { type: String },                // FCM push token (subscribed only)
  8	  type:       { type: String, required: true }, // event type
  9	  path:       { type: String },
 10	  pageType:   { type: String },
 11	  meta:       { type: mongoose.Schema.Types.Mixed }, // event-specific data
 12	  ts:         { type: Date, default: Date.now },
 13	});
 14
 15	// Auto-delete events older than 90 days.
 16	storefrontEventSchema.index({ ts: 1 }, { expireAfterSeconds: 90 * 24 * 60 * 60 });
 17
 18	module.exports = mongoose.model('StorefrontEvent', storefrontEventSchema);
 19

  Notes (observations only):
  * Fields: shopDomain, sessionId, customerId, token, type (required), path,
    pageType, meta (Mixed — arbitrary per-event payload), ts.
  * The `sessionId` comment says "Shopify cart token" but events.js line 138
    and the storefront JS actually put the stable ccfSessionId (a localStorage
    UUID) here — the comment is stale.
  * Indexes: shopDomain, sessionId, customerId each single-field; a TTL index
    on `ts` (expireAfterSeconds 90 days). NO compound { shopDomain, ts } or
    { shopDomain, type } index — the /summary aggregate $match on
    { shopDomain, ts } is served by the shopDomain index alone.
  * events.js line 107 comment says "last 30 days" but this TTL keeps 90 days
    of raw events available.
  * No schema-level validation of `type` — any string is accepted.
  * `meta` shape (from the storefront block) varies by event, e.g.
    product_view -> { productId, title, vendor, type, price, variantId,
    imageUrl }; reached_checkout -> { cartValue, itemCount };
    push_prompt_shown -> { trigger, productId };
    push_prompt_accepted -> { trigger, granted }; page_exit -> { dwellSeconds,
    scrollDepth, pageType }.
