  147  /**
  148   * Returns true if sending a push right now would violate quiet
  149   * hours (10pm-8am) in the given IANA timezone.

  151  function isQuietHours(timezone) {
  152    try {
  153      const now = new Date();
  154      const hour = parseInt(
  155        new Intl.DateTimeFormat('en-US', {
  156          hour: 'numeric',
  157          hour12: false,
  158          timeZone: timezone || 'Asia/Kolkata',
  159        }).format(now),
  160        10
  161      );
  162      return hour >= 22 || hour < 8;
  163    } catch (err) {
  164      console.error('[automation] isQuietHours error:', err.message);
  165      return false; // fail open — don't block sends on a bad timezone
  166    }
  167  }

  169  /**
  170   * Returns true if this cartToken/customerId has already received
  171   * 3+ automation pushes in the last 24 hours.
  172   */
  173  async function isOverFrequencyCap(shopDomain, cartToken, customerId) {
  174    const since = new Date(Date.now() - 24 * 60 * 60 * 1000);
  175    const orClauses = [];
  176    if (cartToken) orClauses.push({ cartToken });
  177    if (customerId) orClauses.push({ customerId });
  178    if (!orClauses.length) return false;
  179  
  180    const count = await ScheduledJob.countDocuments({
  181      shopDomain,
  182      status: 'sent',
  183      sentAt: { $gte: since },
  184      $or: orClauses,
  185    });
  186    return count >= 3;
  187  }

  189  /**
  190   * Polls for due ScheduledJob rows and sends them. Runs every 30s.

  192   * so this is safe even if multiple instances ever run.

  194  async function processScheduledJobs() {
  195    try {
  196      const now = new Date();
  197      const dueJobs = await ScheduledJob.find({
  198        status: 'pending',
  199        runAt: { $lte: now },
  200      }).limit(20);

  202      for (const job of dueJobs) {
  203        // Check quiet hours + frequency cap BEFORE claiming — if
  204        // blocked, leave the job pending and re-check next tick
  205        // (quiet hours) or skip it permanently (frequency cap).
  206        const Store = require('./models/Store');
  207        const store = await Store.findOne({ shopDomain: job.shopDomain }).select('timezone').lean();
  208        const timezone = store?.timezone || 'Asia/Kolkata';
  209  
  210        if (isQuietHours(timezone)) {
  211          console.log(`[automation] Job ${job._id} deferred — quiet hours (${timezone})`);
  212          continue; // leave pending, retry next tick
  213        }
  214  
  215        const overCap = await isOverFrequencyCap(job.shopDomain, job.cartToken, job.customerId);
  216        if (overCap) {
  217          await ScheduledJob.findOneAndUpdate(
  218            { _id: job._id, status: 'pending' },
  219            { status: 'skipped', error: 'Frequency cap reached (3/24h)' }
  220          );
  221          console.log(`[automation] Job ${job._id} skipped — frequency cap reached`);
  222          continue;
  223        }
  224  
  225        // Claim the job — only proceed if we successfully flip it from pending.
  226        const claimed = await ScheduledJob.findOneAndUpdate(
  227          { _id: job._id, status: 'pending' },
  228          { status: 'sent', sentAt: new Date() },
  229          { new: true }
  230        );
  231        if (!claimed) continue; // another process already claimed it
  232  
  233        try {
  234          const channel = job.channel || 'push';
  235          const payload = job.payload || {};
  236

  238            const baseUrl = payload.url || `https://${job.shopDomain}`;
  239            const urlWithJob = baseUrl + (baseUrl.includes('?') ? '&' : '?') + 'ccf_job=' + job._id.toString();

  241              job.shopDomain,
  242              payload.title || 'You left something behind!',
  243              payload.body || 'Come back and check it out.',
  244              urlWithJob,
  245              payload.imageUrl || null,
  246              false,
  247              job.cartToken || null,
  248              false
  249            );
  250  
  251            if (!result.success || result.sent === 0) {
  252              await ScheduledJob.findByIdAndUpdate(job._id, {
  253                status: 'failed',
  254                error: result.error || 'No active subscriber found',
  255              });
  256              console.log(`[automation] Job ${job._id} failed: no subscriber reached`);
  257            } else {
  258              console.log(`[automation] Job ${job._id} sent successfully`);
  259            }
  260          } else if (channel === 'email') {
  261            const customer = await AbandonedCustomer.findOne({
  262              shopDomain: job.shopDomain,
  263              sessionId: job.cartToken,
  264            });
  265            if (!customer || !customer.email) {
  266              await ScheduledJob.findByIdAndUpdate(job._id, {
  267                status: 'skipped',
  268                error: 'No customer email for this cart',
  269              });
  270              console.log(`[automation] Job ${job._id} skipped — no customer email`);
  271            } else {
  272              const cartUrl = `https://${job.shopDomain}/cart`;
  273              const sendResult = await sendAbandonedCartEmail(customer, {
  274                subject: payload.subject || undefined,
  275                body: payload.body || undefined,
  276                cartUrl,
  277              });
  278              if (!sendResult.success) {
  279                await ScheduledJob.findByIdAndUpdate(job._id, {
  280                  status: 'failed',

  286              }
  287            }
  288          }

  291            status: 'failed',

  306  // --- Nightly signal computation — runs at 02:00 server time ------------
  307  function scheduleNightlySignals() {
  308    const now = new Date();
  309    const next2am = new Date();
  310    next2am.setHours(2, 0, 0, 0);
  311    if (next2am <= now) next2am.setDate(next2am.getDate() + 1);
  312    const msUntil2am = next2am - now;
  313    setTimeout(async () => {
  314      try {
  315        const shops = await Store.find({}, 'shopDomain');
  316        for (const s of shops) {
  317          await runNightlySignals(s.shopDomain);
  318        }
  319      } catch (err) {
  320        console.error('[signals] nightly run error:', err.message);
  321      }
  322      scheduleNightlySignals(); // reschedule for next night
  323    }, msUntil2am);
  324  }

  376    await Signal.deleteMany({ shopDomain: shop, profileId, type: { $nin: activeTypes } });
  377  
  378    console.log(`[signals] profile ${profileId}: ${results.length} active`);
  379    return results;
  380  }

  382  async function runNightlySignals(shopDomain) {
  383    const shop = String(shopDomain || '').trim().toLowerCase();
  384    const profiles = await Profile.find(
  385      { shopDomain: shop, suppressed: { $ne: true } },
  386      '_id',
  387      { lean: true }
  388    );
  389    const ids = (profiles || []).map((p) => p._id);
  390  
  391    let processed = 0;
  392    for (let i = 0; i < ids.length; i += 100) {
  393      const batch = ids.slice(i, i + 100);
  394      await Promise.all(
  395        batch.map((id) =>
  396          computeSignalsForProfile(id, shop).catch((err) =>
  397            console.error('[signals] nightly compute error:', err.message)
  398          )
  399        )
  400      );
  401      processed += batch.length;
  402    }
  403  
  404    console.log(`[signals] nightly run complete: ${processed} profiles processed`);
  405    return processed;
  406  }
