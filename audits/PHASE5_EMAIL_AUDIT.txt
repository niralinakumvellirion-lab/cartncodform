 37        return;
 38      }
 39

 41        // Skip entirely if any non-cancelled job already exists for
 42        // this rule+cart (sent or still pending) — avoids re-sending
 43        // step 0 on every subsequent carts/update for a long-lived cart.

 45          shopDomain: shop,
 46          ruleId: rule._id,
 47          cartToken,
 48          status: { $in: ['pending', 'sent'] },

 51          console.log(`[automation] Rule "${rule.name}" already has a job for cart ${cartToken} — skipping re-schedule`);
 52          continue;
 53        }
 54
 55        let cumulativeMinutes = 0;
 56        for (let i = 0; i < rule.steps.length; i++) {
 57          const step = rule.steps[i];
 58          cumulativeMinutes += step.delayMinutes;
 59          const runAt = new Date(Date.now() + cumulativeMinutes * 60 * 1000);
 60
 61          const title = step.title
 62            .replace('{cartValue}', cartContext.cartValue || '')
 63            .replace('{productTitle}', cartContext.firstItemTitle || 'your item');
 64          const body = step.body
 65            .replace('{cartValue}', cartContext.cartValue || '')
 66            .replace('{productTitle}', cartContext.firstItemTitle || 'your item');
 67
 68          try {
 69            await ScheduledJob.create({
 70              shopDomain: shop,
 71              ruleId: rule._id,
 72              stepIndex: i,

174    const count = await ScheduledJob.countDocuments({
175      shopDomain,
176      status: 'sent',
177      sentAt: { $gte: since },
178      $or: orClauses,
179    });
180    return count >= 3;
181  }
182
183  /**
184   * Polls for due ScheduledJob rows and sends them. Runs every 30s.
185   * Uses a claim-update (findOneAndUpdate with status:'pending' filter)
186   * so this is safe even if multiple instances ever run.
187   */
188  async function processScheduledJobs() {
189    try {
190      const now = new Date();
191      const dueJobs = await ScheduledJob.find({
192        status: 'pending',
193        runAt: { $lte: now },
194      }).limit(20);
195
196      for (const job of dueJobs) {
197        // Check quiet hours + frequency cap BEFORE claiming — if
198        // blocked, leave the job pending and re-check next tick
199        // (quiet hours) or skip it permanently (frequency cap).
200        const Store = require('./models/Store');
201        const store = await Store.findOne({ shopDomain: job.shopDomain }).select('timezone').lean();
202        const timezone = store?.timezone || 'Asia/Kolkata';
203
204        if (isQuietHours(timezone)) {
205          console.log(`[automation] Job ${job._id} deferred — quiet hours (${timezone})`);
206          continue; // leave pending, retry next tick
207        }
208
209        const overCap = await isOverFrequencyCap(job.shopDomain, job.cartToken, job.customerId);
210        if (overCap) {
211          await ScheduledJob.findOneAndUpdate(
212            { _id: job._id, status: 'pending' },
213            { status: 'skipped', error: 'Frequency cap reached (3/24h)' }
214          );
215          console.log(`[automation] Job ${job._id} skipped — frequency cap reached`);
216          continue;
217        }
218
219        // Claim the job — only proceed if we successfully flip it from pending.
220        const claimed = await ScheduledJob.findOneAndUpdate(
221          { _id: job._id, status: 'pending' },
222          { status: 'sent', sentAt: new Date() },
223          { new: true }
224        );
225        if (!claimed) continue; // another process already claimed it
226
227        try {
228          const payload = job.payload || {};
229          const baseUrl = payload.url || `https://${job.shopDomain}`;
230          const urlWithJob = baseUrl + (baseUrl.includes('?') ? '&' : '?') + 'ccf_job=' + job._id.toString();
231          const result = await sendPushToCustomers(
232            job.shopDomain,
233            payload.title || 'You left something behind!',
234            payload.body || 'Come back and check it out.',
235            urlWithJob,
236            payload.imageUrl || null,
237            true,
238            job.cartToken || null,
239            false,
240            job.customerId || null
241          );
242
243          if (!result.success || result.sent === 0) {
244            await ScheduledJob.findByIdAndUpdate(job._id, {
245              status: 'failed',
246              error: result.error || 'No active subscriber found',
247            });
248            console.log(`[automation] Job ${job._id} failed: no subscriber reached`);
249          } else {
250            console.log(`[automation] Job ${job._id} sent successfully`);
251          }
252        } catch (err) {
253          await ScheduledJob.findByIdAndUpdate(job._id, {
254            status: 'failed',
255            error: err.message,
256          });
257          console.error(`[automation] Job ${job._id} error:`, err.message);
258        }
259      }
260    } catch (err) {
261      console.error('[automation] processScheduledJobs error:', err.message);
262    }
263  }
264
265  // Poll every 30 seconds.
266  setInterval(processScheduledJobs, 30 * 1000);
267  console.log('[automation] Scheduled job poller started (30s interval)');

   - The send block (lines 227-258) has NO channel branch. It
     ALWAYS calls sendPushToCustomers(...). This is the single
     place to branch: `if (job.channel === 'email') { ...load
     AbandonedCustomer by cartToken, call sendAbandonedCartEmail

   - The claim-update (219-224) flips status pending -> sent
