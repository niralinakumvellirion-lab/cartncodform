   28  function isQuietHour(hour, start, end) {
   29    if (start === end) return false;
   30    if (start < end) return hour >= start && hour < end; // same-day window
   31    return hour >= start || hour < end;                  // wraps midnight (e.g. 22->8)
   32  }

   34  function hourInTz(date, tz) {
   35    try {
   36      const h = parseInt(
   37        new Intl.DateTimeFormat('en-US', {
   38          hour: 'numeric',
   39          hour12: false,
   40          timeZone: tz || 'Asia/Kolkata',
   41        }).format(date),
   42        10
   43      );
   44      return h % 24; // some ICU builds emit "24" at midnight
   45    } catch {
   46      return date.getHours();
   47    }
   48  }

   74  function computeRunAt(profile, quietStart, quietEnd, timezone) {
   75    const now = new Date();
   76    const curHour = hourInTz(now, timezone);
   77  
   78    let targetHour = null;
   79  
   80    const ah = Array.isArray(profile.activeHours)
   81      ? profile.activeHours.filter((h) => Number.isInteger(h) && h >= 0 && h <= 23)
   82      : [];
   83  
   84    if (ah.length >= 3) {
   85      const counts = {};
   86      for (const h of ah) counts[h] = (counts[h] || 0) + 1;
   87      const ranked = Object.keys(counts)
   88        .map(Number)
   89        .filter((h) => !isQuietHour(h, quietStart, quietEnd))
   90        .sort((a, b) => counts[b] - counts[a] || a - b);
   91      if (ranked.length) targetHour = ranked[0];
   92    }
   93  
   94    if (targetHour == null) {
   95      for (let i = 0; i < 24; i++) {
   96        const h = (curHour + 1 + i) % 24;
   97        if (!isQuietHour(h, quietStart, quietEnd)) {
   98          targetHour = h;
   99          break;
  100        }
  101      }
  102      if (targetHour == null) targetHour = (curHour + 1) % 24; // all-quiet fallback
  103    }
  104  
  105    // Next occurrence of targetHour (store-local), aligned to the top of the hour.
  106    let deltaHours = targetHour - curHour;
  107    if (deltaHours <= 0) deltaHours += 24;
  108    const runAt = new Date(now.getTime() + deltaHours * HOUR);
  109    runAt.setMinutes(0, 0, 0);
  110    return runAt;
  111  }

  115  async function runBrainForProfile(profileId, shopDomain) {
  116    const profile = await Profile.findById(profileId);
  117    if (!profile || profile.suppressed) return null;
  118  
  119    const shop = shopDomain || profile.shopDomain;
  120  
  121    const store = await Store.findOne({ shopDomain: shop });
  122    const caps = (store && store.caps) || {};
  123    const perDay = caps.perDay != null ? caps.perDay : 2;
  124    const perWeek = caps.perWeek != null ? caps.perWeek : 5;
  125    const qh = (store && store.quietHours) || {};
  126    const quietStart = qh.start != null ? qh.start : 22;
  127    const quietEnd = qh.end != null ? qh.end : 8;
  128    const timezone = (store && store.timezone) || 'Asia/Kolkata';
  129  
  130    const signals = await Signal.find(
  131      { shopDomain: shop, profileId },
  132      null,
  133      { sort: { strength: -1 } }
  134    );
  135    if (!signals || !signals.length) return null;
  136  
  137    // --- fatigue: daily + weekly caps from the profile's message log ---
  138    const now = Date.now();
  139    const msgs = Array.isArray(profile.messages) ? profile.messages : [];
  140    const sentSince = (ms) =>
  141      msgs.filter((m) => m.sentAt && now - new Date(m.sentAt).getTime() < ms);
  142  
  143    if (sentSince(24 * HOUR).length >= perDay) return null;
  144    if (sentSince(7 * DAY).length >= perWeek) return null;
  145  
  146    // --- rank: strongest first, honouring config + per-signal 6h cooldown ---
  147    let winner = null;
  148    let winnerConfig = null;
  149    for (const s of signals) {
  150      const config = await SignalConfig.findOne({ shopDomain: shop, signalType: s.type });
  151      if (config && config.enabled === false) continue;
  152  
  153      const tooSoon = msgs.some(
  154        (m) => m.type === s.type && m.sentAt && now - new Date(m.sentAt).getTime() < 6 * HOUR
  155      );
  156      if (tooSoon) continue;
  157  
  158      winner = s;
  159      winnerConfig = config;
  160      break;
  161    }
  162    if (!winner) return null;
  163  
  164    // --- channel ---
  165    const channel = selectChannel(winner.type, winnerConfig, profile);
  166    if (!channel) return null;
  167  
  168    // --- send time ---
  169    const runAt = computeRunAt(profile, quietStart, quietEnd, timezone);
  170  
  171    // --- schedule (dedup: one pending job per profile+signal) ---
  172    const existing = await ScheduledJob.findOne({
  173      profileId,
  174      signalType: winner.type,
  175      status: 'pending',
  176    });
  177    if (existing) return null;
  178  
  179    const carts = (profile.identifiers && profile.identifiers.cartTokens) || [];
  180    const strengthStr = Number(winner.strength).toFixed(2);
  181    const reason = `${winner.type} strength=${strengthStr}`;
  182  
  183    try {
  184      await ScheduledJob.create({
  185        shopDomain: shop,
  186        profileId,
  187        signalType: winner.type,
  188        channel,
  189        reason,
  190        payload: {
  191          title: '', // resolved at send time by AI (Phase E)
  192          body: '',  // resolved at send time by AI (Phase E)
  193          url: winner.productId
  194            ? `https://${shop}/products/${winner.productId}`
  195            : `https://${shop}`,
  196          imageUrl: '', // resolved at send time
  197        },
  198        cartToken: carts[0] || null,
  199        ruleId: null,
  200        stepIndex: 0,
  201        status: 'pending',
  202        runAt,
  203      });
  204    } catch (err) {
  205      // 11000 = the unique brain-job index caught a race — treat as already scheduled.
  206      if (err && err.code === 11000) return null;
  207      throw err;
  208    }
  209  
  210    console.log(`[brain] profile ${profileId}: scheduled ${winner.type} on ${channel}`);
  211    return { profileId, signalType: winner.type, channel, runAt, reason };
  212  }

  216  async function runBrainForShop(shopDomain) {
  217    const shop = String(shopDomain || '').trim().toLowerCase();
  218    const profiles = await Profile.find(
  219      { shopDomain: shop, suppressed: { $ne: true } },
  220      '_id',
  221      { lean: true }
  222    );
  223    const ids = (profiles || []).map((p) => p._id);
  224  
  225    let scheduled = 0;
  226    let skipped = 0;
  227  
  228    for (let i = 0; i < ids.length; i += 50) {
  229      const batch = ids.slice(i, i + 50);
  230      const results = await Promise.all(
  231        batch.map((id) =>
  232          runBrainForProfile(id, shop).catch((err) => {
  233            console.error('[brain] profile error:', err.message);
  234            return null;
  235          })
  236        )
  237      );
  238      for (const r of results) {
  239        if (r) scheduled += 1;
  240        else skipped += 1;
  241      }
  242    }
  243  
  244    console.log(`[brain] shop done: ${scheduled} scheduled, ${skipped} skipped`);
  245    return { scheduled, skipped };
