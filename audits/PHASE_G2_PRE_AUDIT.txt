   37      const order = await CodOrder.create({
   38        shopDomain: shopDomain.trim().toLowerCase(),
   39        name: name.trim(),
   40        phone: phone.trim(),
   41        address: address.trim(),
   42        city: (city || '').trim(),
   43        pincode: (pincode || '').trim(),
   44        productName: (productName || '').trim(),
   45        productPrice: Number(productPrice) || 0,
   46        quantity: Number(quantity) || 1,
   47        status: 'pending',
   48      });
   49  
   50      console.log(`[cod] New COD order for ${order.shopDomain} from ${order.name} (${order.phone})`);
   51  
   52      // Identity resolution — the verified COD phone is the strongest identity
   53      // anchor for an Indian D2C shopper. Fire-and-forget.
   54      upsertProfile(order.shopDomain, {
   55        phone: phone,
   56        email: req.body.email || null,
   57      }, {
   58        'channels.email.address': req.body.email || undefined,
   59        'channels.email.source': req.body.email ? 'cod' : undefined,
   60        lastSeenAt: new Date(),
   61      }).catch((err) => console.error('[profile] upsert error:', err.message));
   62  
   63      // Notify the store owner about the new COD order.
   64      const store = await Store.findOne({ shopDomain: order.shopDomain });
   65      const ownerEmail = (store && store.ownerEmail) || process.env.TEST_OWNER_EMAIL;
   66      if (ownerEmail) {
   67        sendNewCodOrderEmail(order, ownerEmail);
   68        console.log(`[cod] New COD order email triggered for owner: ${ownerEmail}`);
   69      } else {
   70        console.log('[cod] No owner email available (set TEST_OWNER_EMAIL) — skipping notification');
   71      }
   72  
   73      // Push-notify the store owner about the new COD order.
   74      sendPushToStore(
   75        order.shopDomain,
   76        '📦 New COD Order',
   77        `${order.name} placed a COD order for ${order.productName}`
   78      );
   79  
   80      return res.status(201).json({ success: true, order });
   81    } catch (err) {
   82      console.error('[cod] POST /order error:', err.message);
   83      return res.status(500).json({ error: 'Failed to save COD order' });
   84    }
   85  });

   87  /**
   88   * PATCH /api/cod/order/:id
   89   * Owner updates the status of a COD order (pending/confirmed/cancelled).
   90   */
   91  router.patch('/order/:id', requireAuth, async (req, res) => {
   92    try {
   93      const { status } = req.body;
   94      const allowed = ['pending', 'confirmed', 'cancelled'];
   95  
   96      if (!allowed.includes(status)) {
   97        return res.status(400).json({ error: `status must be one of ${allowed.join(', ')}` });
   98      }
   99  
  100      const existingOrder = await CodOrder.findById(req.params.id);
  101      if (!existingOrder) {
  102        return res.status(404).json({ error: 'COD order not found' });
  103      }
  104  
  105      if (req.shopDomain !== existingOrder.shopDomain) {
  106        return res.status(403).json({ error: 'Forbidden' });
  107      }
  108  
  109      const order = await CodOrder.findByIdAndUpdate(
  110        req.params.id,
  111        { status },
  112        { new: true }
  113      );
  114  
  115      console.log(`[cod] Order ${order._id} status -> ${status}`);
  116      return res.json({ success: true, order });
  117    } catch (err) {
  118      console.error('[cod] PATCH /order/:id error:', err.message);
  119      return res.status(500).json({ error: 'Failed to update COD order' });
  120    }
  121  });
