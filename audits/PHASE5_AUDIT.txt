  SEND PATTERN — how push actually gets delivered:
   - Two send utilities in backend/utils/pushNotification.js:
       sendPushToStore(shopDomain, title, body)
         -> notifies every PushSubscription row for the store
            (the OWNER's own devices).
       sendPushToCustomers(shopDomain, title, body, url, imageUrl,
                            <customerOnly:true>, cartToken,
                            <broadcast flag>, customerId)
         -> notifies CustomerPushSubscription rows (storefront
            shoppers). Arg 7 (cartToken) narrows to the one cart's
            subscriber; arg 8 true = broadcast to all shop
            subscribers when no cartToken; arg 9 = customerId
            fallback.
   - The AUTOMATION poller (backend/server.js processScheduledJobs)
     calls sendPushToCustomers(job.shopDomain, payload.title,
       payload.body, urlWithJob, payload.imageUrl, true,
       job.cartToken || null, false, job.customerId || null).
     i.e. automation reuses sendPushToCustomers directly, NOT this
     route file. This route file is only the dashboard "manual
     push" surface.
   - Return shape from both utilities: { success, sent, tokensFound?, error? }.

  *** PRE-EXISTING BUG (relevant to Phase 5's auth story) ***
   Lines 48 and 141 still do:
       if (!store || store.ownerEmail !== req.userEmail) return 403;
   But the Shopify-session-token auth swap changed requireAuth to
   set req.shopDomain (NOT req.userEmail). So on these two routes
   req.userEmail is now always undefined -> store.ownerEmail !==
   undefined is always true -> POST /api/push/send and
   POST /api/push/send-customer 403 for EVERY caller. push.js was
   not in the auth-swap's scope and was never updated. Any Phase 5
   work that touches these routes should fix this to
   `req.shopDomain !== <the shop being acted on>` (the pattern
   now used in requireStoreOwner).

 15  async function scheduleCartAbandonJobs(shopDomain, cartToken, customerId, cartContext) {
 16    try {
 17      const shop = String(shopDomain || '').trim().toLowerCase();
 18      if (!cartToken) return;
 19
 20      const rules = await AutomationRule.find({
 21        shopDomain: shop,
 22        trigger: 'cart_abandon',
 23        active: true,
 24      });
 25
 26      if (!rules.length) return;
 27
 28      // Find the subscriber linked to this cart (if any) so we know
 29      // there's someone to notify before scheduling anything.
 30      const sub = await CustomerPushSubscription.findOne({
 31        shopDomain: shop,
 32        cartToken: cartToken,
 33      }).sort({ lastActivityAt: -1 });
 34
 35      if (!sub) {
 36        console.log(`[automation] No subscriber for cartToken ${cartToken} — skipping schedule`);
 37        return;
 38      }
 39
 40      for (const rule of rules) {
 41        // Skip entirely if any non-cancelled job already exists for
 42        // this rule+cart (sent or still pending) — avoids re-sending
 43        // step 0 on every subsequent carts/update for a long-lived cart.
 44        const existingJob = await ScheduledJob.findOne({
 45          shopDomain: shop,
 46          ruleId: rule._id,
 47          cartToken,
 48          status: { $in: ['pending', 'sent'] },
 49        });
 50        if (existingJob) {
 51          console.log(`[automation] Rule "${rule.name}" already has a job for cart ${cartToken} — skipping re-schedule`);
 52          continue;
 53        }
 54
 55        let cumulativeMinutes = 0;
 56        for (let i = 0; i < rule.steps.length; i++) {
 57          const step = rule.steps[i];
 58          cumulativeMinutes += step.delayMinutes;
 59          const runAt = new Date(Date.now() + cumulativeMinutes * 60 * 1000);
 60
 61          const title = step.title
 62            .replace('{cartValue}', cartContext.cartValue || '')
 63            .replace('{productTitle}', cartContext.firstItemTitle || 'your item');
 64          const body = step.body
 65            .replace('{cartValue}', cartContext.cartValue || '')
 66            .replace('{productTitle}', cartContext.firstItemTitle || 'your item');
 67
 68          try {
 69            await ScheduledJob.create({
 70              shopDomain: shop,
 71              ruleId: rule._id,
 72              stepIndex: i,
 73              cartToken,
 74              customerId: customerId || sub.customerId || null,
 75              runAt,
 76              status: 'pending',
 77              payload: {
 78                title,
 79                body,
 80                url: cartContext.productUrl || `https://${shop}`,
 81                imageUrl: step.imageSource === 'product' ? (cartContext.productImageUrl || null) : null,
 82              },
 83            });
 84          } catch (err) {
 85            if (err.code === 11000) {
 86              console.log(`[automation] Duplicate job prevented for rule "${rule.name}", cart ${cartToken}, step ${i}`);
 87            } else {
 88              throw err;
 89            }
 90          }
 91        }
 92
 93        console.log(`[automation] Scheduled ${rule.steps.length} job(s) for rule "${rule.name}" (cart ${cartToken})`);
 94      }
 95    } catch (err) {
 96      console.error('[automation] scheduleCartAbandonJobs error:', err.message);
 97    }

  HOW PUSH STEPS ARE STRUCTURED (for reference to add an email channel):
   - AutomationRule (backend/models/AutomationRule.js):
       shopDomain, name,
       trigger  enum: browse_abandon | cart_abandon | checkout_abandon | back_in_interest
                (only cart_abandon is implemented today),
       conditions (Mixed, unused),
       steps: [ stepSchema ],  active, createdAt.
   - stepSchema (subdoc, { _id: false }):
       delayMinutes  Number, required
       channel       String, enum: ['push'], default 'push'   <-- ONLY 'push' allowed today
       title         String, required
       body          String, required
       imageSource   String, enum: ['product','none'], default 'product'
   - scheduleCartAbandonJobs (above) turns each step into a
     ScheduledJob row:
       { shopDomain, ruleId, stepIndex, cartToken, customerId,
         runAt (now + cumulative delayMinutes), status:'pending',
         payload: { title, body, url, imageUrl } }
     Templating: only two tokens, {cartValue} and {productTitle},
     String.replace (first occurrence only) at schedule time.
   - The poller (backend/server.js processScheduledJobs, 30s tick)
     reads pending ScheduledJob rows and always calls
     sendPushToCustomers(...) — there is NO channel branch. A
     ScheduledJob has no `channel` field; the poller assumes push.

  WHAT AN EMAIL CHANNEL WOULD NEED (derived from the above — not
  a task, just the gap analysis):
   1. stepSchema.channel enum -> add 'email' (and default stays 'push').
   2. ScheduledJob -> add a `channel` field (persist step.channel
      onto the job at schedule time; default 'push' for back-compat).
   3. scheduleCartAbandonJobs -> for an email step, the "is there
      someone to notify?" precheck currently requires a
      CustomerPushSubscription row (lines 30-38). An email step
      should instead require an AbandonedCustomer row WITH an
      email for that cartToken. Right now a cart with an email but
      no push subscription is skipped entirely.
   4. The poller -> branch on job.channel: 'push' ->
      sendPushToCustomers (as today); 'email' -> load
      AbandonedCustomer by { shopDomain, sessionId: job.cartToken }
      and call utils/email.js sendAbandonedCartEmail(row) (or a new
      templated variant using job.payload.title/body).
      Note the poller's quiet-hours + frequency-cap logic is push-
      oriented; decide whether it applies to email.
   5. utils/email.js currently hard-codes subject + HTML and a
      generic FRONTEND_URL CTA — no per-step title/body, no
      per-shop from-address, no EmailTemplate model. A real email
      step wants at least: use step.title as subject, step.body in
      the HTML, and a shop-specific return URL. An EmailTemplate
      model (does not exist) would be the place for per-shop
      branding if that's in scope.
   6. Frontend automation UI (already Polaris-ported in
      frontend/app/admin/page.js — StepEditor / RuleForm) has no
      channel selector; every step is implicitly push.

 47  async function sendAbandonedCartEmail(customer) {
 48    try {
 49      if (!customer || !customer.email) {
 50        // No email on file — nothing to send.
 51        return { success: false, error: 'no customer email' };
 52      }
 53
 54      const items = Array.isArray(customer.cartItems) ? customer.cartItems : [];
 55      const rows = items
 56        .map((i) => {
 57          const qty = i.quantity || 1;
 58          const line = (Number(i.price) || 0) * qty;
 59          return (
 60            `<tr>` +
 61            `<td style="padding:6px 0;">${escapeHtml(i.title || 'Item')} &times; ${qty}</td>` +
 62            `<td style="padding:6px 0;text-align:right;">${money(line)}</td>` +
 63            `</tr>`
 64          );
 65        })
 66        .join('');
 67
 68      const cta = process.env.FRONTEND_URL || '#';
 69
 70      const html = shell(
 71        `<h1 style="font-size:22px;margin:0 0 12px;">You forgot something!</h1>` +
 72          `<p style="margin:0 0 16px;color:#4b5563;">Here's what's still waiting in your cart:</p>` +
 73          `<table style="width:100%;border-collapse:collapse;font-size:14px;">` +
 74          `${rows || '<tr><td style="padding:6px 0;color:#6b7280;">Your saved items</td></tr>'}` +
 75          `<tr><td style="padding:10px 0;border-top:1px solid #e5e7eb;font-weight:700;">Total</td>` +
 76          `<td style="padding:10px 0;border-top:1px solid #e5e7eb;text-align:right;font-weight:700;">` +
 77          `${money(customer.cartValue)}</td></tr>` +
 78          `</table>` +
 79          `<p style="margin:24px 0;">${button(cta, 'Complete Your Order')}</p>` +
 80          `<p style="margin:32px 0 0;font-size:12px;color:#9ca3af;">` +
 81          `This reminder was sent by CartnCodForm</p>`
 82      );
 83
 84      const { error } = await resend.emails.send({
 85        from: FROM,
 86        to: customer.email,
 87        subject: 'You left something behind! 🛒',
 88        html,
 89      });
 90
 91      if (error) throw new Error(error.message || JSON.stringify(error));
 92
 93      console.log(`Email sent: abandoned_cart to ${customer.email}`);
 94      return { success: true };
 95    } catch (err) {
 96      console.error(`Email failed: ${err.message}`);
 97      return { success: false, error: err.message };
 98    }
 99  }
