151  // Wire the four intent triggers. First one to fire wins (session guard).
152  var ccfExitIntentBound = false;
153  function initIntentTriggers() {
154    if (!canPrompt()) return;
155
156    var pageType = getPageType();
157    var meta = pageType === 'product' ? getProductMeta() : null;
158    var pid = meta && meta.productId ? String(meta.productId) : null;
159
160    // Trigger 2 — RETURN VISIT: this productId seen in a previous session.
161    if (pageType === 'product' && pid) {
162      var viewed = [];
163      try { viewed = JSON.parse(localStorage.getItem(VIEWED_KEY) || '[]'); } catch (e) { viewed = []; }
164      if (!Array.isArray(viewed)) viewed = [];
165      if (viewed.indexOf(pid) !== -1) {
166        showSoftPrompt('return_visit', pid);
167      } else {
168        viewed.push(pid);
169        if (viewed.length > 200) viewed = viewed.slice(-200);
170        try { localStorage.setItem(VIEWED_KEY, JSON.stringify(viewed)); } catch (e) {}
171      }
172    }
173
174    // Trigger 1 — DWELL: 45s continuous on a product page (pauses when hidden).
175    if (pageType === 'product') {
176      var dwellRemaining = 10000;
177      var dwellStartedAt = 0;
178      var dwellTimer = null;
179      function startDwell() {
180        if (dwellTimer || dwellRemaining <= 0) return;
181        dwellStartedAt = Date.now();
182        dwellTimer = setTimeout(function () {
183          dwellTimer = null;
184          dwellRemaining = 0;
185          showSoftPrompt('dwell', pid);
186        }, dwellRemaining);
187      }
188      function pauseDwell() {
189        if (!dwellTimer) return;
190        clearTimeout(dwellTimer);
191        dwellTimer = null;
192        dwellRemaining -= (Date.now() - dwellStartedAt);
193        if (dwellRemaining < 0) dwellRemaining = 0;
194      }
195      document.addEventListener('visibilitychange', function () {
196        if (document.visibilityState === 'hidden') pauseDwell();
197        else startDwell();
198      });
199      if (document.visibilityState === 'visible') startDwell();
200    }
201
202    // Trigger 4 — EXIT INTENT: desktop only, mouse toward the browser chrome.
203    if (window.innerWidth > 768 && !ccfExitIntentBound) {
204      ccfExitIntentBound = true;
205      var onExitMove = function (e) {
206        if (e.clientY < 20) {
207          document.removeEventListener('mousemove', onExitMove);
208          showSoftPrompt('exit_intent', null);
209        }
210      };
211      document.addEventListener('mousemove', onExitMove);
212    }
213    // Trigger 3 — ADD TO CART: fired from the cart-add hooks (see trackCart /
214    // the /cart/add.js fetch interceptor), which call showSoftPrompt('add_to_cart').
215  }

--- 1d. Notes on the current prompt (observations only, no changes) --------

  * The card is built entirely in JS with inline `style` attributes — there is
    no <style> block / CSS class. Container id = `ccf-push-prompt`, text id =
    `ccf-prompt-text`, buttons `ccf-allow-btn` / `ccf-deny-btn`.
  * Visual: fixed bottom-right, white, 12px radius, soft shadow, max-width 300px,
    padding 16px, 14px inherit font. Allow = filled indigo (#4f46e5) bold;
    "No thanks" = borderless grey text button.
  * Lifecycle: shows once per session (`ccf_prompt_shown` in sessionStorage via
    markPromptShown / promptAlreadyShown). Auto-dismisses after 8s with no event.
    "No thanks" and a denied native permission both set `ccf_denied_session`.
  * Gate `canPrompt()` requires: Notification API present, permission === 'default',
    not already shown this session, `SUBSCRIBED_KEY` !== '1', no `ccf_denied_session`.
  * DWELL comment on line 174 still says "45s" but the timer constant
    (line 176 `dwellRemaining = 10000`) is 10s — comment is stale after the
    dwell-timeout tweak.
  * The native `Notification.requestPermission()` is only invoked on Allow click
    (line 127) — never on display.
