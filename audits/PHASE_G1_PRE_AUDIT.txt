  244  function RevenueStats({ customers }) {
  245    const stats = useMemo(() => {
  246      const recovered = customers.filter(c => c.status === 'recovered');
  247      const withRevenue = recovered.filter(c =>
  248        typeof c.recoveredRevenue === 'number' && c.recoveredRevenue > 0
  249      );
  250  
  251      const totalRevenue = withRevenue.reduce((sum, c) => sum + c.recoveredRevenue, 0);
  252      const pushAttributed = withRevenue.filter(c => c.attributionSource === 'push');
  253      const pushRevenue = pushAttributed.reduce((sum, c) => sum + c.recoveredRevenue, 0);
  254  
  255      const totalCarts = customers.length;
  256      const recoveryRate = totalCarts > 0
  257        ? ((recovered.length / totalCarts) * 100).toFixed(1)
  258        : '0.0';
  259  
  260      // Determine a display currency — use the most common one found,
  261      // fall back to no symbol if mixed/absent.
  262      const currencies = withRevenue
  263        .map(c => c.recoveredCurrency)
  264        .filter(Boolean);
  265      const currencyCounts = {};
  266      currencies.forEach(c => { currencyCounts[c] = (currencyCounts[c] || 0) + 1; });
  267      const topCurrency = Object.keys(currencyCounts).sort(
  268        (a, b) => currencyCounts[b] - currencyCounts[a]
  269      )[0] || null;
  270      const mixedCurrencies = new Set(currencies).size > 1;
  271  
  272      const revenuePerSend = pushAttributed.length > 0
  273        ? (pushRevenue / pushAttributed.length).toFixed(2)
  274        : null;
  275  
  276      return {
  277        totalRevenue,
  278        pushRevenue,
  279        recoveredCount: recovered.length,
  280        recoveryRate,
  281        totalCarts,
  282        revenuePerSend,
  283        topCurrency,
  284        mixedCurrencies,
  285        hasData: withRevenue.length > 0,
  286      };
  287    }, [customers]);

  360  function CustomerAnalytics({ open, shop, customer, onClose }) {
  361    const [events, setEvents] = useState([]);
  362    const [loading, setLoading] = useState(true);
  363  
  364    useEffect(() => {
  365      if (!customer) return;
  366  
  367      const sessionId = customer.sessionId;
  368      if (!sessionId) {
  369        setLoading(false);
  370        return;
  371      }
  372  
  373      // Step 1: resolve ccfSessionId from cartToken (sessionId)
  374      apiGet(`/api/events/${encodeURIComponent(shop)}/resolve/${encodeURIComponent(sessionId)}`)
  375        .then(function(data) {
  376          const ccfSessionId = data.ccfSessionId;
  377          if (!ccfSessionId) {
  378            // No subscription found for this cart — try direct sessionId lookup
  379            return apiGet(`/api/events/${encodeURIComponent(shop)}/customer/${encodeURIComponent(sessionId)}`);
  380          }
  381          // Step 2: fetch events by stable ccfSessionId
  382          return apiGet(`/api/events/${encodeURIComponent(shop)}/ccfsession/${encodeURIComponent(ccfSessionId)}`);
  383        })
  384        .then(setEvents)
  385        .catch(() => setEvents([]))
  386        .finally(() => setLoading(false));
  387    }, [shop, customer]);
  388  
  389    const iconFor = (type) => ({
  390      page_view: '👁️',
  391      product_view: '🛍️',
  392      collection_view: '📂',
  393      search: '🔍',
  394      cart_view: '🛒',
  395      add_to_cart: '➕',
  396      remove_from_cart: '➖',
  397      cart_update: '✏️',
  398      reached_checkout: '💳',
  399      page_exit: '🚪',
  400    }[type] || '📍');
  401  
  402    return (

  416              renderItem={(e, id) => (
  417                <ResourceItem id={id}>
  418                  <InlineStack gap="300" blockAlign="start" wrap={false}>

    1  // Central place for the backend base URL + a small fetch helper.
    2  export const BACKEND_URL =
    3    process.env.NEXT_PUBLIC_BACKEND_URL || 'http://localhost:4000';
    4  
    5  // fetch() rejects with a bare "TypeError: Failed to fetch" for any network-level
    6  // failure (backend down, wrong port, CORS blocked, DNS). Turn that into a
    7  // message that actually says what to check.
    8  function describeNetworkError(err, method, url) {
    9    const raw = err && err.message ? err.message : String(err);
   10    if (err instanceof TypeError || /failed to fetch|networkerror|load failed/i.test(raw)) {
   11      return (
   12        `Cannot reach the backend (${method} ${url}). ` +
   13        `Is it running at ${BACKEND_URL} and is CORS allowing this origin? ` +
   14        `[${raw}]`
   15      );
   16    }
   17    return raw;
   18  }
   19  
   20  // Auth is a Shopify App Bridge session token. App Bridge is loaded
   21  // synchronously by a plain <script> in the root layout <head>, so by the time
   22  // any api call runs window.shopify is normally already there; the short poll
   23  // below just covers the first render. window.shopify.idToken() returns a
   24  // fresh, short-lived JWT and caches/refreshes internally — no caching here.
   25  async function getAuthToken() {
   26    if (typeof window === 'undefined') {
   27      throw new Error('Not running in browser');
   28    }
   29    let waited = 0;
   30    while (!window.shopify) {
   31      if (waited >= 10000) {
   32        throw new Error('App Bridge did not initialize');
   33      }
   34      await new Promise(resolve => setTimeout(resolve, 100));
   35      waited += 100;
   36    }
   37    return await window.shopify.idToken();
   38  }
   39  
   40  export async function apiGet(path) {
   41    const url = `${BACKEND_URL}${path}`;
   42    const token = await getAuthToken();
   43    const headers = token ? { Authorization: `Bearer ${token}` } : {};
   44  
   45    let res;
   46    try {
   47      res = await fetch(url, { cache: 'no-store', headers });
   48    } catch (err) {
   49      throw new Error(describeNetworkError(err, 'GET', url));
   50    }
   51  
   52    // Retry once with a fresh token on 401 (expired token).
   53    if (res.status === 401 && token) {
   54      const freshToken = await getAuthToken(true);
   55      if (freshToken) {
   56        try {
   57          res = await fetch(url, {
   58            cache: 'no-store',
   59            headers: { Authorization: `Bearer ${freshToken}` },
   60          });
   61        } catch (err) {
   62          throw new Error(describeNetworkError(err, 'GET', url));
   63        }
   64      }
   65    }
   66  
   67    if (!res.ok) {
   68      const body = await res.text().catch(() => '');
   69      throw new Error(`GET ${path} failed (${res.status}): ${body}`);
   70    }
   71    return res.json();
   72  }

   74  export async function apiSend(path, method, body) {
   75    const url = `${BACKEND_URL}${path}`;
   76    const token = await getAuthToken();
   77    const headers = {
   78      'Content-Type': 'application/json',
   79      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   80    };
   81  
   82    let res;
   83    try {
   84      res = await fetch(url, {
   85        method,
   86        headers,
   87        body: JSON.stringify(body),
   88      });
   89    } catch (err) {
   90      throw new Error(describeNetworkError(err, method, url));
   91    }
   92  
   93    // Retry once with a fresh token on 401 (expired token).
   94    if (res.status === 401 && token) {
   95      const freshToken = await getAuthToken(true);
   96      if (freshToken) {
   97        try {
   98          res = await fetch(url, {
   99            method,
  100            headers: {
  101              'Content-Type': 'application/json',
  102              Authorization: `Bearer ${freshToken}`,
  103            },
  104            body: JSON.stringify(body),
  105          });
  106        } catch (err) {
  107          throw new Error(describeNetworkError(err, method, url));
  108        }
  109      }
  110    }
  111  
  112    const data = await res.json().catch(() => ({}));
  113    if (!res.ok) {
  114      throw new Error(data.error || `${method} ${path} failed (${res.status})`);
  115    }
  116    return data;
