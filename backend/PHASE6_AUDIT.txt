   1  const express = require('express');
   2  const router = express.Router();
   3  const path = require('path');
   4  const fs = require('fs');
   5  const crypto = require('crypto');
   6
   7  /**
   8   * Verify Shopify's App Proxy request signature.
   9   * Shopify appends ?signature=<hmac>&<other signed params> to every
  10   * App Proxy request; the HMAC is SHA256 of the remaining params
  11   * sorted by key and concatenated as key=value (no separator),
  12   * keyed with the app's shared secret (SHOPIFY_API_SECRET).
  13   */
  14  function verifyProxySignature(query) {
  15    try {
  16      const { signature, ...rest } = query;
  17      if (!signature) return false;
  18      if (!process.env.SHOPIFY_API_SECRET) {
  19        console.error('[proxy] SHOPIFY_API_SECRET not configured — cannot verify signature');
  20        return false;
  21      }
  22
  23      const sorted = Object.keys(rest)
  24        .sort()
  25        .map(key => {
  26          const val = Array.isArray(rest[key]) ? rest[key].join(',') : rest[key];
  27          return `${key}=${val}`;
  28        })
  29        .join('');
  30
  31      const hash = crypto
  32        .createHmac('sha256', process.env.SHOPIFY_API_SECRET)
  33        .update(sorted)
  34        .digest('hex');
  35
  36      return hash === signature;
  37    } catch (err) {
  38      console.error('[proxy] Signature verification error:', err.message);
  39      return false;
  40    }
  41  }
  42
  43  // Serve Firebase SW via App Proxy
  44  // URL: https://cartncod-form.myshopify.com/apps/cartncodform/sw.js
  45  router.get('/sw.js', (req, res) => {
  46    if (!verifyProxySignature(req.query)) {
  47      console.warn('[proxy] Invalid or missing App Proxy signature on /sw.js — allowing for now (soft enforcement)');
  48      // return res.status(403).send('Invalid signature'); // uncomment once confirmed safe
  49    }
  50
  51    const swPath = path.join(__dirname, '../public/cartncodform-sw.js');
  52
  53    if (!fs.existsSync(swPath)) {
  54      return res.status(404).send('Service worker not found');
  55    }
  56
  57    const swContent = fs.readFileSync(swPath, 'utf8');
  58
  59    res.setHeader('Content-Type', 'application/javascript; charset=utf-8');
  60    res.setHeader('Service-Worker-Allowed', '/apps/cartncodform/');
  61    res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
  62    res.setHeader('Pragma', 'no-cache');
  63    res.setHeader('Expires', '0');
  64    res.setHeader('Access-Control-Allow-Origin', '*');
  65    res.setHeader('X-Content-Type-Options', 'nosniff');
  66
  67    console.log('[proxy] Served SW, headers set for root scope');
  68    res.send(swContent);
  69  });
  70
  71  // Health check
  72  router.get('/health', (req, res) => {

  74      console.warn('[proxy] Invalid or missing App Proxy signature on /health — allowing for now (soft enforcement)');

  76    }
  77
  78    res.json({ success: true, service: 'CartnCodForm Proxy' });

OBSERVATIONS (not changes — context for Phase 6 planning)
================================================================================

A. COD form frontend (cod-form/[shop]/page.js)
   - Public storefront page. Route param `[shop]` -> `params.shop`
     (decodeURIComponent). `productName` + `price` come from query string.
   - Client component, wrapped in <Suspense> because of useSearchParams()
     (App Router CSR-bailout rule — same pattern as admin/page.js).
   - Submits via `apiSend('/api/cod/order', 'POST', {...})`.
     NOTE: apiSend() calls getAuthToken() -> window.shopify.idToken(). This
     page is NOT embedded in Shopify Admin (it's a storefront page), so
     window.shopify will never exist here -> getAuthToken() will spin for
     10s then throw "App Bridge did not initialize", and the POST is never
     sent. This path is currently BROKEN for the standalone storefront form.
     (POST /api/cod/order itself is a public, no-auth route, so the fix is to
     not send an Authorization header from this page — e.g. a plain fetch, or
     an apiSend variant that skips the token.)
   - Fields collected: name, phone, address, city, pincode, quantity.
     No email field on the form. No cart-token / sessionId linkage — a COD
     order is not tied to an AbandonedCustomer row.
   - Tailwind utility classes reference custom colors `bg-brand` /
     `bg-brand-dark` (defined in tailwind.config) + a local `.input` class
     via styled-jsx global.

B. backend/routes/cod.js
   - POST /api/cod/order — PUBLIC (no requireAuth). Validates shopDomain,
     name, phone, address. Creates CodOrder (status 'pending'). Fires (fire
     and forget, not awaited):
       * sendNewCodOrderEmail(order, ownerEmail)  — ownerEmail from
         Store.ownerEmail, else process.env.TEST_OWNER_EMAIL.
       * sendPushToStore(shopDomain, title, body) — owner FCM push.
   - PATCH /api/cod/order/:id — requireAuth. Validates status in
     [pending,confirmed,cancelled], loads order, then authz check:
         if (!store || store.ownerEmail !== req.userEmail) -> 403
     >>> STALE AUTH BUG: after the Shopify session-token swap, requireAuth no
     longer sets req.userEmail (tokens carry no email). req.userEmail is
     always undefined here, so:
         - store has an ownerEmail  -> 'someone@x' !== undefined -> 403 ALWAYS
           (owners can never update COD order status from the embedded admin)
         - store.ownerEmail falsy   -> undefined !== undefined -> false ->
           passes with NO shop check at all (weak).
     Same class of bug that was already fixed in routes/push.js /send and
     /send-customer. The correct fix mirrors those:
         const store = await Store.findOne({ shopDomain: existingOrder.shopDomain });
         if (!store || req.shopDomain !== existingOrder.shopDomain) return 403;
     (existingOrder.shopDomain is already normalized lowercase by the schema.)
   - Frontend caller: admin/page.js StoreView.updateOrderStatus() ->
     apiSend(`/api/cod/order/${id}`, 'PATCH', { status }). That call DOES run
     inside the embedded admin, so it will carry a valid session token and
     set req.shopDomain — only the server-side check is wrong.

C. backend/routes/proxy.js
   - Mounted at app.use('/apps/cartncodform', proxyRouter) in server.js.
   - verifyProxySignature() implements Shopify App Proxy HMAC (SHA256 of
     sorted key=value params, no separator, keyed with SHOPIFY_API_SECRET).
     NOTE: this is the App-Proxy scheme (concatenated, no '&'), distinct from
     the query-string HMAC used for OAuth/redirect verification.
   - BOTH routes only WARN on a bad/missing signature — the 403 returns are
     commented out ("soft enforcement"). So /apps/cartncodform/sw.js and
     /apps/cartncodform/health are effectively open.
   - /sw.js reads backend/public/cartncodform-sw.js from disk and serves it
     with wide-open CORS + Service-Worker-Allowed: /apps/cartncodform/ +
     no-cache headers. 404 if the file is missing.
   - No COD-form-serving route here — the proxy currently only serves the SW
     and a health check. If Phase 6 is about serving the COD form through the
     App Proxy (so it renders on the merchant's storefront domain), that
     route does not exist yet.

D. backend/models/CodOrder.js
   - Flat schema: shopDomain (indexed, lowercased), name, phone, address
     (all required), city, pincode, productName, productPrice (default 0),
     quantity (default 1, min 1), status enum [pending,confirmed,cancelled]
     default pending, createdAt.
   - No email field. No sessionId / cartToken. No updatedAt. No compound
     index (only the single shopDomain index).

E. Cross-cutting
   - server.js mounts:  /api/cod -> codRoutes ,  /apps/cartncodform ->
     proxyRouter  (confirmed in earlier reads of server.js).
   - The only auth helper imported by cod.js is requireAuth; requireStoreOwner
     is NOT used here (cod.js does its own inline store/owner check — the
     buggy one in B).
