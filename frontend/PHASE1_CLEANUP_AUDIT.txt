153  /**
154   * Returns true if this cartToken/customerId has already received
155   * 3+ automation pushes in the last 24 hours.
156   */
157  async function isOverFrequencyCap(shopDomain, cartToken, customerId) {
158    const since = new Date(Date.now() - 24 * 60 * 60 * 1000);
159    const orClauses = [];
160    if (cartToken) orClauses.push({ cartToken });
161    if (customerId) orClauses.push({ customerId });
162    if (!orClauses.length) return false;
163
164    const count = await ScheduledJob.countDocuments({
165      shopDomain,
166      status: 'sent',
167      sentAt: { $gte: since },
168      $or: orClauses,
169    });
170    return count >= 3;
171  }

173  /**
174   * Polls for due ScheduledJob rows and sends them. Runs every 30s.
175   * Uses a claim-update (findOneAndUpdate with status:'pending' filter)
176   * so this is safe even if multiple instances ever run.
177   */
178  async function processScheduledJobs() {
179    try {
180      const now = new Date();
181      const dueJobs = await ScheduledJob.find({
182        status: 'pending',
183        runAt: { $lte: now },
184      }).limit(20);
185
186      for (const job of dueJobs) {
187        // Check quiet hours + frequency cap BEFORE claiming — if
188        // blocked, leave the job pending and re-check next tick
189        // (quiet hours) or skip it permanently (frequency cap).
190        const Store = require('./models/Store');
191        const store = await Store.findOne({ shopDomain: job.shopDomain }).select('timezone').lean();
192        const timezone = store?.timezone || 'Asia/Kolkata';
193
194        if (isQuietHours(timezone)) {
195          console.log(`[automation] Job ${job._id} deferred — quiet hours (${timezone})`);
196          continue; // leave pending, retry next tick
197        }
198
199        const overCap = await isOverFrequencyCap(job.shopDomain, job.cartToken, job.customerId);
200        if (overCap) {
201          await ScheduledJob.findOneAndUpdate(
202            { _id: job._id, status: 'pending' },
203            { status: 'skipped', error: 'Frequency cap reached (3/24h)' }
204          );
205          console.log(`[automation] Job ${job._id} skipped — frequency cap reached`);
206          continue;
207        }
208
209        // Claim the job — only proceed if we successfully flip it from pending.
210        const claimed = await ScheduledJob.findOneAndUpdate(
211          { _id: job._id, status: 'pending' },
212          { status: 'sent', sentAt: new Date() },
213          { new: true }
214        );
215        if (!claimed) continue; // another process already claimed it
216
217        try {
218          const payload = job.payload || {};
219          const baseUrl = payload.url || `https://${job.shopDomain}`;
220          const urlWithJob = baseUrl + (baseUrl.includes('?') ? '&' : '?') + 'ccf_job=' + job._id.toString();
221          const result = await sendPushToCustomers(
222            job.shopDomain,
223            payload.title || 'You left something behind!',
224            payload.body || 'Come back and check it out.',
225            urlWithJob,
226            payload.imageUrl || null,
227            true,
228            job.cartToken || null,
229            false,
230            job.customerId || null
231          );
232
233          if (!result.success || result.sent === 0) {
234            await ScheduledJob.findByIdAndUpdate(job._id, {
235              status: 'failed',
236              error: result.error || 'No active subscriber found',
237            });
238            console.log(`[automation] Job ${job._id} failed: no subscriber reached`);
239          } else {
240            console.log(`[automation] Job ${job._id} sent successfully`);
241          }
242        } catch (err) {
243          await ScheduledJob.findByIdAndUpdate(job._id, {
244            status: 'failed',
245            error: err.message,
246          });
247          console.error(`[automation] Job ${job._id} error:`, err.message);
248        }
249      }
250    } catch (err) {
251      console.error('[automation] processScheduledJobs error:', err.message);
252    }
253  }
